cd clientRegistrationPolicy && ./scripts/generate-root-ca.sh && ./scripts/generate-client-jwks.sh

docker compose up -d --build

>>Definir contenidos para las variables en ssa.env

SSA_ISSUER=https://directorio-qa.finanzasabiertas.cl
SSA_SOFTWARE_ID=MATCH-BCI
SSA_ORGANISATION_ID="MATCH-BCI"
SSA_SOFTWARE_JWKS_URI=https://fintech-match-bci.localtest.me/.well-known/jwks.json
SSA_SOFTWARE_CLIENT_NAME="SFA mTLS POC - DCR + SSA"
SSA_SOFTWARE_VERSION=1.2.9
SSA_REDIRECT_URIS=http://localhost:3000/callback
SSA_JWT_KID=sfa-poc-directory-1

>>COMPILAR PROVIDER KEYCLOAK (una vez o tras cambios en src/)

./scripts/build-provider.sh

>>GENERAR SSA PARA REGISTRO DE FINTECH
cd clientRegistrationPolicy
SSA=$(./scripts/generate-test-ssa.sh)

>> REGISTRO DCR + SSA (gateway mTLS :8443)
curl --cert client-jwks/MATCH-BCI/transport.crt \
     --key client-jwks/MATCH-BCI/transport.key \
     --cacert registro_participantes/ca/root-ca.crt \
     --location 'https://sfa.localtest.me:8443/realms/sfa-mtls-poc/clients-registrations/openid-connect' \
     --header 'Content-Type: application/json' \
     --header "X-Software-Statement: $SSA" \
     --data '{
    "grant_types": ["authorization_code", "refresh_token", "client_credentials"],
    "response_types": ["code"],
    "token_endpoint_auth_method": "tls_client_auth",
    "tls_client_auth_subject_dn": "C=CL,O=MATCH-BCI,CN=MATCH-BCI",
    "tls_client_certificate_bound_access_tokens": true,
    "redirect_uris": ["http://localhost:3000/callback"],
    "scope": "accounts:read"
  }'

>> TOKEN client_credentials (certificado X509, sin client_assertion)
curl --cert client-jwks/MATCH-BCI/transport.crt \
     --key client-jwks/MATCH-BCI/transport.key \
     --cacert registro_participantes/ca/root-ca.crt \
     --location 'https://sfa.localtest.me:8443/realms/sfa-mtls-poc/protocol/openid-connect/token' \
     --header 'Content-Type: application/x-www-form-urlencoded' \
     --data-urlencode 'grant_type=client_credentials' \
     --data-urlencode 'client_id=MATCH-BCI' \
     --data-urlencode 'scope=accounts:read'

>> API Resource Server (gateway mTLS :9443)
curl --cert client-jwks/MATCH-BCI/transport.crt \
     --key client-jwks/MATCH-BCI/transport.key \
     --cacert registro_participantes/ca/root-ca.crt \
     https://api.localtest.me:9443/health
